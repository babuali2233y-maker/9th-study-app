import React from 'react';
import { studyData } from '../data/studyData';

function ChapterView({ subject, onSelectChapter, onStartTest, onBack }) {
  const chapters = studyData.chapters[subject.id] || [];

  return (
    <div className="chapter-view-container">
      <div className="header-section">
        <button className="back-btn" onClick={onBack}>← Back</button>
        <h2>{subject.name} - Chapters</h2>
      </div>

      <div className="chapters-list">
        {chapters.map((chapter) => (
          <div key={chapter.id} className="chapter-card">
            <div className="chapter-info">
              <h3>{chapter.title}</h3>
              <p>📖 Study Material • 🎯 {chapter.mcqs.length} MCQs • ❓ {chapter.questions.length} Questions</p>
            </div>
            <div className="chapter-actions">
              <button 
                className="btn btn-study"
                onClick={() => onSelectChapter(chapter)}
              >
                Study
              </button>
              <button 
                className="btn btn-test"
                onClick={() => {
                  onSelectChapter(chapter);
                  onStartTest();
                }}
              >
                Test
              </button>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}

export default ChapterView;
