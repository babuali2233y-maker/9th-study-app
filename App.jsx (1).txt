import React, { useState } from 'react';
import './App.css';
import SubjectList from './components/SubjectList';
import ChapterView from './components/ChapterView';
import TestMode from './components/TestMode';
import Results from './components/Results';

function App() {
  const [currentView, setCurrentView] = useState('subjects'); // subjects, chapters, study, test, results
  const [selectedSubject, setSelectedSubject] = useState(null);
  const [selectedChapter, setSelectedChapter] = useState(null);
  const [testResults, setTestResults] = useState(null);

  const handleSubjectSelect = (subject) => {
    setSelectedSubject(subject);
    setCurrentView('chapters');
  };

  const handleChapterSelect = (chapter) => {
    setSelectedChapter(chapter);
    setCurrentView('study');
  };

  const handleStartTest = () => {
    setCurrentView('test');
  };

  const handleTestComplete = (results) => {
    setTestResults(results);
    setCurrentView('results');
  };

  const handleBack = () => {
    if (currentView === 'chapters' || currentView === 'study' || currentView === 'test' || currentView === 'results') {
      setCurrentView('subjects');
      setSelectedSubject(null);
      setSelectedChapter(null);
    }
  };

  return (
    <div className="app">
      <header className="app-header">
        <h1>📚 9th Class Sindh</h1>
        <p>G•H & Junaid | made by</p>
      </header>

      <main className="app-main">
        {currentView === 'subjects' && (
          <SubjectList onSelectSubject={handleSubjectSelect} />
        )}

        {currentView === 'chapters' && (
          <ChapterView 
            subject={selectedSubject}
            onSelectChapter={handleChapterSelect}
            onStartTest={handleStartTest}
            onBack={handleBack}
          />
        )}

        {currentView === 'study' && (
          <div>
            <button className="back-btn" onClick={() => setCurrentView('chapters')}>← Back</button>
            <div className="study-content">
              <h2>{selectedChapter.title}</h2>
              <div className="study-text">
                {selectedChapter.content}
              </div>
              <div className="study-questions">
                <h3>Questions</h3>
                <ol>
                  {selectedChapter.questions.map((q, idx) => (
                    <li key={idx} className="question-item">
                      <p><strong>{q.question}</strong></p>
                      <p className="answer">Answer: {q.answer}</p>
                    </li>
                  ))}
                </ol>
              </div>
            </div>
          </div>
        )}

        {currentView === 'test' && (
          <TestMode 
            subject={selectedSubject}
            chapter={selectedChapter}
            onComplete={handleTestComplete}
            onBack={() => setCurrentView('chapters')}
          />
        )}

        {currentView === 'results' && (
          <Results 
            results={testResults}
            chapter={selectedChapter}
            onBack={() => setCurrentView('chapters')}
          />
        )}
      </main>

      <footer className="app-footer">
        <p>Study Hard, Score High! 💪</p>
      </footer>
    </div>
  );
}

export default App;
